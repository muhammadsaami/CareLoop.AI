# Phase 3 - RAG retrieval (ChromaDB)
# 0.5.15 is the last 0.5.x line; it works with the pinned pydantic 2.6.1 and

chromadb==0.5.15

# Phase 3 - local semantic embeddings.
#
# app/rag/embeddings/semantic.py imports onnxruntime and tokenizers DIRECTLY,
# so they are declared explicitly rather than inherited from chromadb: relying
# on a transitive pin means a chromadb upgrade could silently remove the
# inference runtime out from under the retrieval service.
#
# onnxruntime runs the model on CPU. This is the whole point of the choice:
# it avoids torch (~2.5 GB installed) for a 33M-parameter model, and
# onnxruntime is already present because chromadb depends on it, so the
# semantic provider adds no heavyweight runtime. Inference is local - no
# embedding API, no API key, and no outbound call on the request path.
onnxruntime>=1.17.0
tokenizers>=0.15.0

# Used ONLY by the one-time operator CLI
# (`python -m app.rag.embeddings.fetch_model`) to download the model weights.
# The application never calls it while serving a request.
huggingface_hub>=0.23.0

# chromadb 0.5.15 calls posthog.capture(user_id, event, properties) with three
# POSITIONAL args.  posthog 4+ changed the signature to capture(event, **kwargs),
# so 4+ raises TypeError on every telemetry event.  Telemetry is disabled in
# code (anonymized_telemetry=False) and could never carry document text - only
# the chroma version and three infrastructure settings - but the pin keeps the
# logs clean and keeps the library consistent if it is ever enabled.
posthog>=2.4.0,<4

# ── Phase 4 — LangGraph grounded-answer agent ────────────────────────────────
# langgraph ONLY.  `langchain` is deliberately NOT a dependency: the agent
# calls the existing app.llm.base.LLMProvider abstraction, so nothing imports
# langchain at runtime.
#
# Pinned to 1.2.11, which requires langchain-core>=1.4.7,<2 - already satisfied
# by the chromadb-transitive langchain-core 1.6.0.  The agent imports no
# langchain-community/-text-splitters/-openai/-groq package, so the
# pre-existing langchain 0.3.x vs core 1.x conflict in the development
# environment is neither triggered nor altered by this line.  Adding it
# upgrades no existing package.
langgraph==1.2.11

# --- Phase 5 -- Scheduling and notifications -----------------------------------
# The reminder engine runs in a Celery worker; Redis is the broker and the
# result backend.  Both are declared explicitly rather than inherited, so a
# transitive upgrade cannot remove the queue the scheduler depends on.
#
# celery 5.6.x is pinned because 6.x changes task configuration defaults.
# redis 5.1.1 is pinned as the client celery and the app's own caching layer
# both use it; redis-py 6.x drops support for the event-loop patterns celery
# 5.6 relies on.
#
# The API process imports the Celery app but never connects: `celery_app.py` is
# reached from the worker entry point, not from `app.main`, so the API boots
# with no broker running.
celery==5.6.3
redis==5.1.1

# Zoneinfo database for `app.core.timezones`.  On Windows, `zoneinfo` resolves
# names through the OS, which ships no tzdata by default, so an IANA name like
# 'Asia/Kolkata' would raise at reminder-creation time.  The package is pinned
# so DST behaviour is identical on every platform - and identical in CI, which
# is not optional when a wrong offset shifts a dose by hours.
#
# Pinned to the version the suite was verified against; moving it forward
# requires re-running tests/test_timezones.py, which is the guard for exactly
# this risk.
tzdata==2026.1
